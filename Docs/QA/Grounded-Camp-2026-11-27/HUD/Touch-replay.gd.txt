func screen(id):

 RenderingServer.force_draw(false)
 var frame=root.get_texture().get_image()
 frame.save_jpg(out_dir.path_join(id+".jpg"),.94)
 captured[id]={"tick":Engine.get_physics_frames(),"size":str(frame.get_size()),"hp":field.player_hp,"hint":hud.prompt.text,"cam":field.get_zoom_ratio(),"wood_units":int(hud.numbers.wood.text),"flights":field.carry.flights.size()}

 field=load("res://field_opening.tscn").instantiate()
 root.add_child(field)
 current_scene=field
 camera=field.camera

   got_run=true
   screen("steer")

   screen("pay")
   got_transfer=true
  if field.inventory.complete("center_hearth") and field.camp_active:break
 steer(Vector2.ZERO)
 for i in 55:await physics_frame
 checks["paid by approaching"]=field.inventory.complete("center_hearth") and got_transfer

 touch(5,mid-Vector2(60,0),true)
 touch(6,mid+Vector2(60,0),true)
 for i in 10:
  for finger in [5,6]:
   var drag=InputEventScreenDrag.new()
   drag.index=finger
   drag.position=mid+Vector2(60.+float(i+1)*6.,0)*(-1 if finger==5 else 1)
   root.push_input(drag,true)
  await physics_frame
 touch(5,mid-Vector2(120,0),false)
 touch(6,mid+Vector2(120,0),false)
 for i in 12:await physics_frame
 screen("pinch")
 checks["pinch"]=field.get_zoom_ratio()>1.8
 var q=field.graphics.quality_button
 touch(9,q.get_global_rect().get_center(),true)

 await physics_frame
 checks["quality toggles without travel"]=not field.graphics.ultra and field.touch_id<0
 var data={"checks":checks,"capture":captured,"viewport":str(root.size),"canvas":str(safe),"renderer":RenderingServer.get_video_adapter_name()}
 var file=FileAccess.open(out_dir.path_join("replay.json"),FileAccess.WRITE);file.store_string(JSON.stringify(data,"  ")+"\n");file.close()
 await Profile.drain_field_audio(self,field)
 field.queue_free()

 quit(1 if false in checks.values() else 0)
